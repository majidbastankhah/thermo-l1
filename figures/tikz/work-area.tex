  % the area under the path, cut into thin strips of width dV: each strip is
  % one small piece of work, P_ext dV, and together they make up |W|
  \pgfmathsetmacro{\N}{14}
  \foreach \i in {0,...,13} {
    \pgfmathsetmacro{\xa}{1.3 + \i*(5.6-1.3)/14}
    \pgfmathsetmacro{\xb}{1.3 + (\i+1)*(5.6-1.3)/14}
    \pgfmathsetmacro{\xm}{(\xa+\xb)/2}
    \pgfmathsetmacro{\ym}{4.6/\xm*1.1}
    \ifodd\i
      \fill[duBlue!22] (\xa,0) rectangle (\xb,\ym);
    \else
      \fill[duPurple!20] (\xa,0) rectangle (\xb,\ym);
    \fi
    \draw[white, line width=0.6pt] (\xb,0) -- (\xb,\ym);
  }
  % one strip picked out
  \pgfmathsetmacro{\xa}{1.3 + 6*(5.6-1.3)/14}
  \pgfmathsetmacro{\xb}{1.3 + 7*(5.6-1.3)/14}
  \pgfmathsetmacro{\ym}{4.6/((\xa+\xb)/2)*1.1}
  \fill[duPurple!70] (\xa,0) rectangle (\xb,\ym);
  \draw[path1, arrowhead=0.28] plot[domain=5.6:1.3, samples=80] (\x, {4.6/\x*1.1});

  \node[statedot] at (1.3,{4.6/1.3*1.1}) {};
  \node[statedot] at (5.6,{4.6/5.6*1.1}) {};
  \node[lbl, above right=0pt and 1pt] at (1.3,{4.6/1.3*1.1}) {2};
  \node[lbl, above right=0pt and 1pt] at (5.6,{4.6/5.6*1.1}) {1};

  % labels for the picked-out strip
  \draw[duGrey, line width=0.6pt] ({(\xa+\xb)/2},{\ym+0.05}) -- (3.95,2.75);
  \node[lbl, anchor=west, align=left] at (3.95,2.9)
    {one strip: $P_\text{ext}\,\mathrm{d}V$};
  \draw[{Stealth[length=2mm]}-{Stealth[length=2mm]}, line width=0.7pt]
    (\xa,-0.32) -- (\xb,-0.32) node[midway, below, font=\small] {$\mathrm{d}V$};

  \node[lbl, duPurple, anchor=west, align=left, fill=white, fill opacity=0.85, text opacity=1, inner sep=2pt] at (3.9,4.05)
    {sum of all strips\\$\big|W\big| = \displaystyle\int P_\text{ext}\,\mathrm{d}V$};

  \draw[dashed, duGrey, line width=0.56pt] (1.3,0) -- (1.3,{4.6/1.3*1.1});

  \node[slbl, below] at (5.6,0) {$V_1$};
