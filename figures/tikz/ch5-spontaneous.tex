\newcommand{\pxs}{0.0125}   % cm per pixel
\newcommand{\panel}[6]{%  #1 x0, #2 x1, #3 y0, #4 y1 (pixels, y down), #5 X, #6 Y (cm)
  \begin{scope}
    \clip (#5,#6) rectangle ++({(#2-#1)*\pxs},{(#4-#3)*\pxs});
    \node[anchor=south west, inner sep=0] at ({#5-#1*\pxs},{#6-(514-#4)*\pxs})
      {\includegraphics[width=8.6375cm]{../img/ch5-spontaneous-img.png}};
  \end{scope}}
\newcommand{\tick}[1]{\draw[duGreen!80!black, line width=1.6pt, line cap=round,
  line join=round] ($#1+(-0.18,0)$) -- ++(0.14,-0.16) -- ++(0.32,0.4);}
\newcommand{\cross}[1]{\draw[duRed, line width=1.6pt, line cap=round]
  ($#1+(-0.16,-0.16)$) -- ++(0.32,0.32) ($#1+(-0.16,0.16)$) -- ++(0.32,-0.32);}
